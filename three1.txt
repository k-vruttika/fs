import React from "react";

class EdRow extends React.Component {
  render() {
    return (
      <tr>
        <td>{this.props.id}</td>
        <td>{this.props.name}</td>
        <td>{this.props.place}</td>
        <td>{this.props.stddeg}</td>
        <td>{this.props.grade}</td>
      </tr>
    );
  }
}

class EdTable extends React.Component {
  render() {
    return (
      <table>
        <tbody>
          <tr><th>id</th><th>name</th><th>place</th><th>education</th><th>grade</th></tr>
          <EdRow id="1" name="National Public School" place="Bengaluru" stddeg="10th" grade="A+" />
          <EdRow id="2" name="Indian Institute of Science" place="Bengaluru" stddeg="under grade" grade="9.8" />
        </tbody>
      </table>
    );
  }
}

class Education extends React.Component {
  render() {
    return <EdTable />;
  }
}

export default Education;
