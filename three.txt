>Education.js:
import React from "react";

class EdRow extends React.Component {
  render() {
    const { id, name, place, stddeg, grade } = this.props;
    return <tr><td>{id}</td><td>{name}</td><td>{place}</td><td>{stddeg}</td><td>{grade}</td></tr>;
  }
}

class EdTable extends React.Component {
  render() {
    const education = [
      { id: 1, name: "National Public School", place: "Bengaluru", stddeg: "10th", grade: "A+" },
      { id: 2, name: "Indian Institute of Science", place: "Bengaluru", stddeg: "under grade", grade: "9.8" }
    ];

    return (
      <table>
        <thead><tr><th>id</th><th>name</th><th>place</th><th>education</th><th>grade</th></tr></thead>
        <tbody>{education.map(e => <EdRow key={e.id} {...e} />)}</tbody>
      </table>
    );
  }
}

class Education extends React.Component {
  render() {
    return <EdTable />;
  }
}

export default Education;
