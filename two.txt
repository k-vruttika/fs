server.js:
const express = require('express');
const app = express();

app.use(express.static(__dirname));
app.use(express.urlencoded({ extended: true }));

const page = (d, m) => `
    <h2>Student Information (${m})</h2>
    Name: <b>${d.name}</b><br>
    Branch: <u>${d.branch}</u><br>
    Semester: ${d.semester}<br><br>
    <a href="/">Go Back</a>`;

app.get('/get', (req, res) => res.send(page(req.query, 'GET')));
app.post('/post', (req, res) => res.send(page(req.body, 'POST')));

app.listen(3001, () => console.log('Server running on port 3001'));
